# tonal 6.12.0 API

Every public function of the `tonal` npm package, generated from its
TypeScript source. One entry per function: the signature, then a one-line
description, then an example when there is one. The types that functions
return are listed after the functions of each module.

Conventions:
- All functions are pure. Nothing mutates its arguments.
- Invalid input does not throw: functions return an object with
  `empty: true`, an empty string `""`, an empty array, or null/undefined
  (see each signature). Check for it.
- Notes are strings like "C4", "Eb5", "F#" (a note without octave is a pitch
  class). Intervals are strings like "3M", "5P", "-2m" ("M3" and "P5" are also
  accepted).
- Functions marked [deprecated] still work but must not be used in new code.

Guide for agents: https://tonaljs.github.io/tonal/docs/agents.md
Index: https://tonaljs.github.io/tonal/llms.txt

## Top-level functions

import { note, interval, transpose, distance } from "tonal";

accToAlt(acc: string): number
  Get the alteration number of an accidentals string.
  accToAlt("##") // => 2
  accToAlt("bb") // => -2
  accToAlt("") // => 0

altToAcc(alt: number): string
  Get the accidentals string of an alteration number.
  altToAcc(2) // => "##"
  altToAcc(-1) // => "b"
  altToAcc(0) // => ""

chroma({ step, alt }: Pitch): number
  Get the chroma (pitch class number, from 0 to 11) of a pitch.
  chroma({ step: 0, alt: 1 }) // => 1

coordinates(pitch: Pitch): PitchCoordinates
  Get the coordinates of a pitch object: `[fifths]` for a pitch class and `[fifths, octaves]` for a note or an interval (the direction is applied to both numbers).
  coordinates({ step: 4, alt: 0, oct: 4 }) // => [1, 4]
  coordinates({ step: 4, alt: 0 }) // => [1]

coordToInterval(coord: PitchCoordinates, forceDescending?: boolean): Interval
  Get the Interval object from its coordinates (`[fifths, octaves]`).
  coordToInterval([1, 0]).name // => "5P"
  coordToInterval([-1, 0]).name // => "-5P"

coordToNote(noteCoord: PitchCoordinates): Note
  Get the Note object from its coordinates: `[fifths]` for a pitch class or `[fifths, octaves]` for a note.
  coordToNote([1, 0]).name // => "G0"
  coordToNote([0, 4]).name // => "C4"

distance(fromNote: NoteLiteral, toNote: NoteLiteral): IntervalName
  Get the interval name between two notes or pitch classes, or "" if either is not valid.
  distance("C4", "G4") // => "5P"
  distance("C", "E") // => "3M"
  distance("G4", "C4") // => "-5P"

height({ step, alt, oct, dir }: Pitch): number
  Get the height of a pitch: its distance in semitones from C0 (so it is the midi number minus 12). A descending interval has a negative height, and a pitch without octave gets a number far below any note.
  height({ step: 0, alt: 0, oct: 4 }) // => 48
  height({ step: 0, alt: 0 }) // => -1200

interval(src: IntervalLiteral): Interval
  Get the Interval object of an interval name ("5P" or "P5"), a pitch object or an object with a `name`. Invalid input returns an object with `empty: true` and `name: ""`.
  interval("5P").semitones // => 7
  interval("P5").name // => "5P"
  interval("m3").type // => "majorable"

isNamed(this: unknown, ...args: unknown[]): boolean
  [deprecated] Use isNamedPitch instead.

isNamedPitch(src: unknown): src is NamedPitch
  Check if a value is an object with a string `name` property.
  isNamedPitch({ name: "C4" }) // => true
  isNamedPitch("C4") // => false

isPitch(pitch: unknown): pitch is Pitch
  Check if a value is an object with numeric `step` and `alt` properties.
  isPitch({ step: 0, alt: 0 }) // => true
  isPitch({ name: "C" }) // => false

midi(pitch: Pitch): number | null
  Get the midi number of a pitch, or `null` if the pitch has no octave or is outside the midi range (0 to 127).
  midi({ step: 0, alt: 0, oct: 4 }) // => 60
  midi({ step: 0, alt: 0 }) // => null

note(src: NoteLiteral): Note
  Get the Note object of a note literal (a note name, a pitch object or an object with a `name`). Pitch classes ("Bb") and notes with octave ("Bb4") are valid. Invalid input returns an object with `empty: true` and `name: ""`.
  note("Bb4") // => { name: "Bb4", pc: "Bb", midi: 70, chroma: 10, ... }
  note("Bb4").freq // => 466.1637615180899
  note("x").empty // => true

pitch(coord: PitchCoordinates): Pitch
  Get the pitch object from its coordinates (the inverse of `coordinates`).
  pitch([1]) // => { step: 4, alt: 0 }
  pitch([1, 0]) // => { step: 4, alt: 0, oct: 0 }

stepToLetter(step: number): string
  Get the note letter of a step (0 is C, 6 is B), or "" if the step is not valid.
  stepToLetter(0) // => "C"
  stepToLetter(6) // => "B"

tokenizeInterval(str?: IntervalName): IntervalTokens
  Split an interval name into `[number, quality]` (both strings). It accepts the number first ("3M") or the quality first ("M3"). Anything that doesn't match gives `["", ""]`.
  tokenizeInterval("3M") // => ["3", "M"]
  tokenizeInterval("M3") // => ["3", "M"]
  tokenizeInterval("-5P") // => ["-5", "P"]

tokenizeNote(str: string): NoteTokens
  Split a note name into `[letter, accidentals, octave, rest]` (all strings, the letter in uppercase). Anything that doesn't match gives empty strings.
  tokenizeNote("Bb4") // => ["B", "b", "4", ""]
  tokenizeNote("C# major") // => ["C", "#", "", "major"]

tonicIntervalsTransposer(intervals: string[], tonic: string | undefined | null): (normalized: number) => '' | NoteName
  Create a function that, given a (possibly negative) scale degree index, returns the note of that degree: the intervals are cycled and the octave goes up (or down) every time they wrap. It returns "" for every degree if there is no tonic.
  const degree = tonicIntervalsTransposer(["1P", "3M", "5P"], "C");
  [0, 1, 2, 3, -1].map(degree) // => ["C", "E", "G", "C", "G"]

transpose(noteName: NoteLiteral, intervalName: IntervalLiteral | [number, number]): NoteName
  Transpose a note by an interval.
  transpose("C4", "5P") // => "G4"
  transpose("D", "3M") // => "F#"
  transpose("x", "3M") // => ""

type Direction = 1 | -1

type Interval = {
  empty: boolean;
  name: IntervalName;
  num: number;
  q: Quality;
  type: Type;
  step: number;
  alt: number;
  dir: Direction;
  simple: number;

}

type IntervalLiteral = Pitch | NamedPitch | IntervalName

type IntervalName = KnownIntervalName | (string & {})
  An interval name. Any string is accepted, quality first too ("M3"); editors suggest the common names (`KnownIntervalName`): every interval up to 15, up or down, number first as tonal writes them, such as "3M" or "-5P".

type NamedPitch = {
  name: string;
}

type Note = {
  empty: boolean;
  name: NoteName;
  letter: string;
  acc: string;
  pc: PcName;
  chroma: number;
  height: number;
  coord: PitchCoordinates;
  midi: number | null;
  freq: number | null;
  step: number;
  alt: number;
  oct?: number | undefined;
  dir?: Direction | undefined;
}

type NoteLiteral = Pitch | NamedPitch | NoteName

type NoteName = (string & {}) | KnownNoteName
  A note name, with or without octave. Any string is accepted; editors suggest the common names (`KnownNoteName`): "C" to "B" with nothing, "#", "b", "##" or "bb", alone or at octaves 0 to 8, such as "Eb" or "C4".

type PcName = (string & {}) | KnownPcName
  A pitch class name (a note without octave). Any string is accepted; editors suggest the 35 common names (`KnownPcName`): "C" to "B" with nothing, "#", "b", "##" or "bb", such as "C" or "Eb".

type Pitch = {
  step: number;
  alt: number;
  oct?: number | undefined;
  dir?: Direction | undefined;
}

type PitchCoordinates = PitchClassCoordinates | NoteCoordinates | IntervalCoordinates

type Quality = 'm' | 'M' | 'P' | `A${string}` | `d${string}`

type Type = 'perfectable' | 'majorable'

## AbcNotation

import { AbcNotation } from "tonal"; // or: import * as AbcNotation from "@tonaljs/abc-notation";

AbcNotation.abcToScientificNotation(str: string): string
  Convert a note in ABC notation into a note in scientific notation. It returns "" if not valid.
  AbcNotation.abcToScientificNotation("c") // => "C5"
  AbcNotation.abcToScientificNotation("_B,") // => "Bb3"

AbcNotation.distance(from: string, to: string): string
  Get the interval name between two ABC notes, or "" if either is not valid.
  AbcNotation.distance("C", "G") // => "5P"
  AbcNotation.distance("C", "c") // => "8P"

AbcNotation.scientificToAbcNotation(str: string): string
  Convert a note in scientific notation into a note in ABC notation. It returns "" if not valid or if the note has no octave.
  AbcNotation.scientificToAbcNotation("C#4") // => "^C"
  AbcNotation.scientificToAbcNotation("Bb3") // => "_B,"

AbcNotation.tokenize(str: string): AbcTokens
  Split an ABC note into `[accidentals, letter, octave marks]`. Anything that is not an ABC note gives `["", "", ""]`.
  AbcNotation.tokenize("^c'") // => ["^", "c", "'"]
  AbcNotation.tokenize("x") // => ["", "", ""]

AbcNotation.transpose(note: string, interval: string): string
  Transpose an ABC note by an interval. It returns an ABC note, or "" if not valid.
  AbcNotation.transpose("C", "5P") // => "G"
  AbcNotation.transpose("c", "3M") // => "e"

## Array

import { Array } from "tonal"; // or: import * as Array from "@tonaljs/array";

Array.compact(arr: any[]): any[]
  Get a copy of the array without the falsy values, except 0 (so null, undefined, "", false and NaN are removed).
  Array.compact(["a", null, "b", 0, undefined]) // => ["a", "b", 0]

Array.permutations(arr: any[]): any[]
  Get all the permutations of an array (n! arrays).
  Array.permutations(["a", "b"]) // => [["a", "b"], ["b", "a"]]
  Array.permutations(["a", "b", "c"]) // =>
  [

Array.range(from: number, to: number): number[]
  Create a list of numbers from `from` to `to`, both included, in steps of one (descending if `to` is lower). It returns [] if either is not finite.
  Array.range(-2, 2) // => [-2, -1, 0, 1, 2]
  Array.range(2, -2) // => [2, 1, 0, -1, -2]
  Array.range(1.5, 4) // => [1.5, 2.5, 3.5]

Array.rotate<T>(times: number, arr: T[]): T[]
  Rotate a list: move the first `times` items to the end (a negative number rotates the other way). It returns a new array.
  Array.rotate(1, [1, 2, 3]) // => [2, 3, 1]
  Array.rotate(-1, [1, 2, 3]) // => [3, 1, 2]

Array.shuffle(arr: any[], rnd?: () => number): any[]
  Shuffle the array in place (Fisher–Yates) and return it. Pass a `rnd` function (returning numbers from 0 to 1) to make it deterministic; `Math.random` by default.
  Array.shuffle(["C", "D", "E", "F"], () => 0.5) // => ["C", "F", "D", "E"]

Array.sortedNoteNames(notes: string[]): string[]
  Get the names of the valid notes sorted from lowest to highest. Pitch classes are listed before notes, and anything that is not a note is removed.
  Array.sortedNoteNames(["c2", "c5", "c1", "c0", "c6", "c"])
  Array.sortedNoteNames(["c", "F", "G", "a", "b", "h", "J"])

Array.sortedUniqNoteNames(arr: string[]): string[]
  Same as sortedNoteNames, but with repeated names removed.
  Array.sortedUniqNoteNames(["a", "b", "c2", "1p", "p2", "c2", "b", "c", "c3"])

## Chord

import { Chord } from "tonal"; // or: import * as Chord from "@tonaljs/chord";

Chord.chord(src: ChordNameOrTokens): Chord
  An alias of `Chord.get`, handy when importing from `@tonaljs/chord`.
  Chord.chord("Dm7").symbol; // => "Dm7"

Chord.chordScales(name: string): string[]
  Get the names of the scale types that contain all the notes of a chord, as type names without tonic. It returns `[]` if the chord isn't valid.
  Chord.chordScales("C7b9"); // => ["phrygian dominant", "flamenco", "phrygian b4", "mixolydian b2", "spanish heptatonic", "half-whole diminished", "chromatic"]

Chord.degreeOf(chordName: ChordNameOrTokens, noteName: NoteName): ChordDegreeName
  Get the degree name that a note plays in a chord, counted from the major scale of the root (the bass of a slash chord doesn't change it). The opposite of `Chord.noteOf`.
  Chord.degreeOf("C7b9", "Db"); // => "b9"
  Chord.degreeOf("C7b9", "C#4"); // => "b9"
  Chord.degreeOf("Cm", "Eb"); // => "b3"

Chord.degrees(chordName: ChordNameOrTokens, tonic?: string): (degree: number) => '' | NoteName
  Get a function that returns the note of a chord by its position, 1 being the tonic. The position counts the chord notes, so the 3rd note of C is "G" and the 4th is "C" an octave up. For the note of the degree named "3", see `Chord.noteOf`; for positions starting at 0, `Chord.steps`.
  [1, 2, 3, 4].map(Chord.degrees("C")); // => ["C", "E", "G", "C"]
  [1, 2, 3, 4].map(Chord.degrees(["C4", "M"])); // => ["C4", "E4", "G4", "C5"]
  Chord.degrees("Cmaj7")(4); // => "B"

Chord.degreesByImportance(chordName: ChordNameOrTokens): ChordDegreeName[]
  Get the degree names of a chord (as in `Chord.formula`) sorted by how much they define it, to choose which ones to keep when there are fewer voices than chord tones. It returns `[]` if the chord isn't valid.
  Chord.degreesByImportance("C13"); // => ["3", "b7", "13", "1", "9", "5"]
  Chord.degreesByImportance("Cm7b5"); // => ["b3", "b7", "b5", "1"]
  Chord.degreesByImportance("C7#5"); // => ["3", "b7", "#5", "1"]

Chord.detect(source: string[], options?: Partial<DetectOptions>): string[]
  Find the chord symbols that have the notes of a list, best first. It returns `[]` if no chord matches or no note is valid.
  Chord.detect(["D", "F#", "A", "C"]); // => ["D7"]
  Chord.detect(["E4", "G4", "C5"]); // => ["CM/E", "Em#5"]
  Chord.detect(["C", "E", "B"]); // => ["Cmaj7no5"]

Chord.extended(chordName: string): string[]
  Get the symbols of the chords that have all the notes of a chord and at least one more, on the same tonic. It returns `[]` if the chord isn't valid. The opposite of `Chord.reduced`.
  Chord.extended("Cmaj7"); // => ["Cmaj#4", "Cmaj7#9", "Cmaj7#9#11", "Cmaj9", "Cmaj13", "CM7add13", "Cmaj9#11", "CM13#11", "Cmaj11", "CM7b9"]

Chord.formula(chordName: ChordNameOrTokens): ChordDegreeName[]
  Get the formula of a chord: its degree names, in the order of its intervals, counted from the major scale of the root. It returns `[]` if the chord isn't valid.
  Chord.formula("Cm7"); // => ["1", "b3", "5", "b7"]
  Chord.formula("dim7"); // => ["1", "b3", "b5", "bb7"]
  Chord.formula("C13/E"); // => ["1", "3", "5", "b7", "9", "13"]

Chord.get(src: ChordNameOrTokens): Chord
  Get a chord from its symbol, as an object with `notes`, `intervals`, `symbol`, `name`, `tonic`, `bass`, `quality` and more. The symbol can have a tonic and a bass (`"C7/E"`), and the chord can be given as tokens `[tonic, type, bass]`. Without a tonic, the chord has intervals but no notes. It returns `{ empty: true, ... }` if the symbol isn't valid.
  Chord.get("Cmaj7").notes; // => ["C", "E", "G", "B"]
  Chord.get("C7/E").notes; // => ["E", "G", "Bb", "C"]
  Chord.get(["C", "maj7"]).symbol; // => "Cmaj7"

Chord.getChord(typeName: string, optionalTonic?: string, optionalBass?: string): Chord
  Get a chord from a chord type name, an optional tonic and an optional bass. It returns the same object as `Chord.get`, and `{ empty: true, ... }` if the type, the tonic or the bass isn't valid. Unlike `Chord.get`, the type isn't parsed from a symbol: `Chord.getChord("maj7", "C")` is `Chord.get("Cmaj7")`.
  Chord.getChord("maj7", "C").notes; // => ["C", "E", "G", "B"]
  Chord.getChord("maj7", "C", "E").notes; // => ["E", "G", "B", "C"]
  Chord.getChord("maj7", "C", "E").symbol; // => "Cmaj7/E"

Chord.noteOf(chordName: ChordNameOrTokens, degree: ChordDegreeName): NoteName
  Get the note of a chord degree name as a pitch class, spelled by transposing the root. The opposite of `Chord.degreeOf`. The degree is counted from the major scale of the root, as in chord formulas: the third of Cm7 is "b3", and its "3" is "E".
  Chord.noteOf("C7b9", "b9"); // => "Db"
  Chord.noteOf("C#7", "b9"); // => "D"
  Chord.noteOf("Cm7", "b3"); // => "Eb"

Chord.notes(chordName: ChordNameOrTokens, tonic?: string): string[]
  Get the notes of a chord, the same as `Chord.get(name).notes` (a slash chord starts at its bass). The optional `tonic` replaces the chord's own tonic, so a chord type name alone works, and a tonic with an octave gives notes with octaves. It returns `[]` if the chord isn't valid or has no tonic.
  Chord.notes("Cmaj7"); // => ["C", "E", "G", "B"]
  Chord.notes("Cmaj7", "C4"); // => ["C4", "E4", "G4", "B4"]
  Chord.notes("maj7", "D"); // => ["D", "F#", "A", "C#"]

Chord.reduced(chordName: string): string[]
  Get the symbols of the chords whose notes are all in a chord and are fewer, on the same tonic. It returns `[]` if the chord isn't valid. The opposite of `Chord.extended`.
  Chord.reduced("Cmaj7"); // => ["C5", "CM"]
  Chord.reduced("Cmaj9"); // => ["C5", "CM", "Cmaj7", "Csus2", "CMadd9"]

Chord.steps(chordName: ChordNameOrTokens, tonic?: string): (normalized: number) => '' | NoteName
  Get a function that returns the note of a chord by its position, 0 being the tonic (like `Chord.degrees` but 0-based). Negative steps go down. The function returns `""` if the chord isn't valid.
  [0, 1, 2, 3].map(Chord.steps("C")); // => ["C", "E", "G", "C"]
  [-1, 0, 1].map(Chord.steps(["C4", "M"])); // => ["G3", "C4", "E4"]

Chord.tokenize(name: string): ChordNameTokens
  Split a chord name into `[tonic, type, bass]`. Each part is a string that is `""` when missing, and a name without tonic is all type. It never throws.
  Chord.tokenize("Cmaj7"); // => ["C", "maj7", ""]
  Chord.tokenize("Cmaj7/E"); // => ["C", "maj7", "E"]
  Chord.tokenize("mMaj7"); // => ["", "mMaj7", ""]

Chord.transpose(chordName: string, interval: string): string
  Transpose a chord symbol by an interval, keeping its type and transposing its tonic and bass. It returns the symbol as is if it has no tonic (or isn't a chord), so it doesn't check that the type is valid.
  Chord.transpose("Dm7", "P4"); // => "Gm7"
  Chord.transpose("C/E", "M2"); // => "D/F#"
  Chord.transpose("nonsense", "P4"); // => "nonsense"

## ChordType

import { ChordType } from "tonal"; // or: import * as ChordType from "@tonaljs/chord-type";

ChordType.add(intervals: string[], aliases: string[], fullName?: string): void
  Add a chord type to the dictionary, so `ChordType.get` and `Chord.get` know it. It changes the dictionary for the whole program.
  ChordType.add(["1P", "5P", "7m"], ["7no3"], "dominant seventh no third");
  Chord.get("C7no3").notes; // => ["C", "G", "Bb"]

ChordType.addAlias(chord: ChordType, alias: string): void
  Add an alias (another symbol) to a chord type of the dictionary, so `ChordType.get` and `Chord.get` accept it. It changes the dictionary for the whole program.
  ChordType.addAlias(ChordType.get("maj7"), "Δ7");
  Chord.get("CΔ7").notes; // => ["C", "E", "G", "B"]

ChordType.all(): ChordType[]
  Get a copy of the list of all chord types in the dictionary.
  ChordType.all().length; // => 115
  ChordType.all()[0].name; // => "fifth"

ChordType.chordType(type: ChordTypeName): ChordType
  [deprecated] Use ChordType.get instead.

ChordType.complexity(intervals: string[]): number
  Get how complex a chord type is from its interval names (`ChordType.get(name).complexity` has it for listed types), to sort chord lists from the plainest to the most colourful. It's the number of notes added to a triad, plus two for each chromatic note:
  ChordType.complexity(["1P", "3M", "5P", "7M", "9M"]); // => 2
  ChordType.get("m7b5").complexity; // => 1
  ChordType.get("dim7").complexity; // => 3

ChordType.entries(): ChordType[]
  [deprecated] Use ChordType.all instead.

ChordType.family(intervals: string[]): ChordFamily
  Get the family of a chord type from its interval names (`ChordType.get(name).family` has it for listed types). It returns `""` for no intervals.
  ChordType.family(["1P", "3M", "5P", "7m"]); // => "dominant"
  ChordType.family(["1P", "4P", "5P", "7m"]); // => "suspended"

ChordType.get(type: ChordTypeName): ChordType
  Get a chord type (a chord without tonic) by its name, alias, chroma or set number. It returns an empty chord type (`{ empty: true, ... }`) if it's not in the dictionary. The type has `name`, `aliases`, `intervals`, `quality` and `family`. To get a chord with notes, use `Chord.get`.
  ChordType.get("major").name; // => "major"
  ChordType.get("maj7").intervals; // => ["1P", "3M", "5P", "7M"]
  ChordType.get("M").aliases; // => ["M", "^", "", "maj"]

ChordType.keys(): string[]
  Get every key that `ChordType.get` accepts: names, aliases, chromas and set numbers (as strings).
  ChordType.keys().slice(0, 6); // => ["2", "4", "5", "6", "7", "9"]

ChordType.names(): string[]
  Get the long names of all the chord types in the dictionary, in the dictionary order. Types without long name are left out.
  ChordType.names().slice(0, 4); // => ["fifth", "suspended fourth", "suspended fourth seventh", "augmented"]

ChordType.removeAll(): void
  Remove every chord type from the dictionary, to define your own with `ChordType.add`. It changes the dictionary for the whole program.

ChordType.symbols(): string[]
  Get the main symbol (the first alias) of every chord type in the dictionary, in the dictionary order.
  ChordType.symbols().slice(0, 4); // => ["5", "M7#5sus4", "7#5sus4", "sus4"]

Collection.compact(arr: any[]): any[]
  Get a copy of the collection without the falsy values, except 0 (so null, undefined, "", false and NaN are removed).

  [

  Create a list of numbers from `from` to `to`, both included, in steps of one (descending if `to` is lower). It returns [] if either is not finite.
  Collection.range(-2, 2) // => [-2, -1, 0, 1, 2]

  Collection.shuffle(["C", "D", "E", "F"], () => 0.5) // => ["C", "F", "D", "E"]

## Interval

import { Interval } from "tonal"; // or: import * as Interval from "@tonaljs/interval";

Interval.abs(name: IntervalName): IntervalName
  Get the ascending version of an interval: the interval itself if it ascends, and its negation if it descends.
  Interval.abs("-3M") // => "3M"
  Interval.abs("3M") // => "3M"
  Interval.abs("1d") // => "1A" (one semitone down, like "-1A")

Interval.add(a: IntervalName, b: IntervalName): IntervalName | undefined
  Add two intervals. It returns the interval name, or "" if either interval is not valid.
  Interval.add("3m", "5P") // => "7m"
  Interval.add("3M", "3M") // => "5A"

Interval.addTo(interval: string): (other: string) => IntervalName | undefined
  Get a function that adds the given interval to another interval.
  ["1P", "2M", "3M"].map(Interval.addTo("5P")) // => ["5P", "6M", "7M"]

Interval.consonance(name: IntervalName, options?: ConsonanceOptions): Consonance | ''
  Classify an interval as a perfect consonance (unison, fifth and octave), an imperfect consonance (thirds and sixths) or a dissonance (seconds, sevenths, the fourth, and every augmented or diminished interval).
  Interval.consonance("5P") // => "perfect"
  Interval.consonance("10m") // => "imperfect"
  Interval.consonance("-6M") // => "imperfect"

Interval.distance(fromNote: NoteLiteral, toNote: NoteLiteral): IntervalName
  Get the interval name between two notes or pitch classes, or "" if either is not valid.
  Interval.distance("C4", "G4") // => "5P"
  Interval.distance("G4", "C4") // => "-5P"
  Interval.distance("C", "E") // => "3M"

Interval.fromSemitones(semitones: number): IntervalName
  Get interval name from semitones number. Since there are several interval names for the same number, the name it's arbitrary, but deterministic.
  Interval.fromSemitones(7) // => "5P"
  Interval.fromSemitones(-7) // => "-5P"

Interval.get(src: IntervalLiteral): Interval
  Get the Interval object (name, number, quality, semitones...) of an interval name ("4P" or "P4"). Invalid input gives an object with `empty: true` and `name: ""`.
  Interval.get("x").empty // => true
  Interval.get("P4").semitones // => 5
  Interval.get("P4").name // => "4P"

Interval.invert(name: IntervalName): IntervalName
  Get the inversion (https://en.wikipedia.org/wiki/Inversion_(music)#Intervals) of an interval: the interval that completes its simplified version to an octave. A unison and an octave invert to each other, compound intervals are simplified first, and the direction is kept.
  Interval.invert("3m") // => "6M"
  Interval.invert("2M") // => "7m"
  Interval.invert("1P") // => "8P"

Interval.isConsonant(name: IntervalName, options?: ConsonanceOptions): boolean
  Is the interval a perfect or an imperfect consonance? See `consonance`.
  Interval.isConsonant("6M") // => true
  Interval.isConsonant("-12P") // => true
  Interval.isConsonant("2M") // => false

Interval.multiply(name: IntervalName, times: number): IntervalName
  Multiply an interval by an integer: stack it that many times. A negative number stacks it in the opposite direction, and zero gives a unison.
  Interval.multiply("5P", 3) // => "13M"
  Interval.multiply("2M", 3) // => "4A"
  Interval.multiply("3M", -2) // => "-5A"

Interval.name(name: string): IntervalName
  Get the normalized name of an interval (number first), or "" if not valid.
  Interval.name("4P") // => "4P"
  Interval.name("P4") // => "4P"
  Interval.name("C4") // => ""

Interval.names(): IntervalName[]
  Get the names of the natural intervals: unison, major second and third, perfect fourth and fifth, and minor sixth and seventh.
  Interval.names() // => ["1P", "2M", "3M", "4P", "5P", "6m", "7m"]

Interval.negate(name: IntervalName): IntervalName
  Get the same interval in the opposite direction.
  Interval.negate("3M") // => "-3M"
  Interval.negate("-5P") // => "5P"
  Interval.negate("1A") // => "-1A"

Interval.num(name: string): number
  Get the number of an interval (4 for a fourth; negative if descending), or NaN if not valid.
  Interval.num("P4") // => 4
  Interval.num("-9M") // => -9

Interval.pitchClass(name: IntervalName): IntervalName
  Get the pitch class of an interval: the interval within an octave, and ascending by letter, that leads to the same pitch class. It ignores octaves and direction, like the distance between two pitch classes.
  Interval.pitchClass("10M") // => "3M"
  Interval.pitchClass("-3M") // => "6m"
  Interval.pitchClass("8P") // => "1P"

Interval.quality(name: string): 'm' | 'M' | 'P' | `A${string}` | `d${string}`
  Get the quality of an interval: "P", "M", "m", "A" (or "AA"...), "d" (or "dd"...). It is "" if not valid.
  Interval.quality("P4") // => "P"
  Interval.quality("3m") // => "m"

Interval.semitones(name: string): number
  Get the size of an interval in semitones (negative if descending), or NaN if not valid.
  Interval.semitones("P4") // => 5
  Interval.semitones("-3M") // => -4

Interval.simplify(name: IntervalName): IntervalName
  Get the simplified version of an interval: the same quality and direction, with the number reduced to an octave or less. An octave stays an octave, so only a unison simplifies to a unison.
  Interval.simplify("9M") // => "2M"
  Interval.simplify("2M") // => "2M"
  Interval.simplify("-2M") // => "-2M"

Interval.subtract(a: IntervalName, b: IntervalName): IntervalName | undefined
  Subtract the second interval from the first. It returns the interval name, or "" if either interval is not valid.
  Interval.subtract('5P', '3M') // => '3m'
  Interval.subtract('3M', '5P') // => '-3m'

Interval.transposeFifths(interval: IntervalName, fifths: number): IntervalName
  Move an interval a number of perfect fifths up (or down, if negative), keeping its octave. It returns "" if the interval is not valid.
  Interval.transposeFifths("3M", 1) // => "7M"
  Interval.transposeFifths("5P", -1) // => "1P"

## Midi

import { Midi } from "tonal"; // or: import * as Midi from "@tonaljs/midi";

Midi.chroma(midi: number): number
  Get the pitch class of a midi number, from 0 up to (not including) 12. A decimal midi number keeps its fraction.
  Midi.chroma(61) // => 1
  Midi.chroma(61.5) // => 1.5
  Midi.chroma(-1) // => 11

Midi.freqToMidi(freq: number): number
  Get the midi number of a frequency in hertz. The midi number can contain decimals (rounded to two digits).
  Midi.freqToMidi(220) // => 57
  Midi.freqToMidi(261.62) // => 60
  Midi.freqToMidi(261) // => 59.96

Midi.isMidi(arg: any): arg is Midi | `${number}`
  Check if a value is a valid midi number: a number, or a numeric string, from 0 to 127. Note names are not midi numbers. A numeric string stays a string: convert it with `Midi.toMidi` (or `+value`) before using it as a number.
  Midi.isMidi(60) // => true
  Midi.isMidi("60") // => true
  Midi.isMidi(128) // => false

Midi.midiToFreq(midi: number, tuning?: number): number
  Get the frequency in hertz of a midi number.
  Midi.midiToFreq(69) // => 440
  Midi.midiToFreq(69, 432) // => 432

Midi.midiToNoteName(midi: number, options?: ToNoteNameOptions): string
  Get the note name of a midi number. Altered notes use flats unless `sharps: true`, and `pitchClass: true` drops the octave. Decimals are rounded to the nearest note, and NaN or infinite numbers give "".
  Midi.midiToNoteName(61) // => "Db4"
  Midi.midiToNoteName(61, { pitchClass: true }) // => "Db"
  Midi.midiToNoteName(61, { sharps: true }) // => "C#4"

Midi.pcset(notes: number[] | string): number[]
  Given a list of midi numbers, returns the pitch class set: the unique pitch classes (whole numbers from 0 to 11), sorted. Decimal midi numbers are rounded to the nearest note first, and numbers that aren't finite are ignored.
  Midi.pcset([62, 63, 60, 65, 70, 72]) // => [0, 2, 3, 5, 10]
  Midi.pcset([60.4, 64, 67.6]) // => [0, 4, 8]
  Midi.pcset("100100100101") // => [0, 3, 6, 9, 11]

Midi.pcsetDegrees(notes: number[] | string, tonic: number): (degree: number) => number | undefined
  Same as pcsetSteps, but degrees start at 1 (the tonic) and go down to -1 (the note below the tonic). The function returns undefined for degree 0.
  const degrees = Midi.pcsetDegrees([0, 2, 4], 60);
  [-1, 1, 2, 3, 4].map(degrees) // => [52, 60, 62, 64, 72]

Midi.pcsetNearest(notes: number[] | string): (midi: number) => number | undefined
  Get a function that maps a midi number to the nearest midi number whose pitch class is in the set (a tie goes up). Can be used to constrain a note to a scale. The function returns undefined if the number is not finite.
  const nearest = Midi.pcsetNearest(Scale.get("D dorian").chroma);
  [60, 61, 62, 63, 64, 65, 66].map(nearest) // => [60, 62, 62, 63, 65, 65, 67]

Midi.pcsetSteps(notes: number[] | string, tonic: number): (step: number) => number
  Get a function that maps a step number to a midi number: step 0 is the tonic (the first pitch class of the set above it), step 1 the second, and so on, going up an octave every time the set wraps. Negative steps go down.
  const steps = Midi.pcsetSteps(Scale.get("D dorian").chroma, 60);
  [-2, -1, 0, 1, 2, 3].map(steps) // => [57, 58, 60, 62, 63, 65]

Midi.toMidi(note: NoteName | number): number | null
  Get the midi number (0 to 127) of a note name or midi number. It returns null if the note is not valid, has no octave or is out of range.
  Midi.toMidi("C4") // => 60
  Midi.toMidi(60) // => 60
  Midi.toMidi("60") // => 60

  Get the note name of a midi number, using sharps for altered notes. Decimals are rounded to the nearest note, and a non-numeric value gives "".

## Pcset

import { Pcset } from "tonal"; // or: import * as Pcset from "@tonaljs/pcset";

Pcset.chroma(set: Set): string
  Get the chroma of a pitch class set: a 12-character string of "0" and "1", one per pitch class starting from C. It's `"000000000000"` for an invalid set.
  Pcset.chroma(["c", "d", "e"]); //=> "101010000000"

Pcset.chromas(): PcsetChroma[]
  Get all the possible pitch class sets that contain C, as chromas: the 2048 chromas from `"100000000000"` to `"111111111111"`. To have them on another note you have to rotate them.
  Pcset.chromas().length; // => 2048
  Pcset.chromas().slice(0, 2); // => ["100000000000", "100000000001"]

Pcset.filter(set: Set): (notes: NoteName[]) => NoteName[]
  Get a function that keeps the notes of a list that belong to a pitch class set, ignoring octaves. The notes are returned as they were given.
  Pcset.filter(["C", "D", "E"])(["c2", "c#2", "d2", "c3", "c#3", "d3"]); // => ["c2", "d2", "c3", "d3"]
  Pcset.filter(["C2"])(["c2", "c#2", "d2", "c3", "c#3", "d3"]); // => ["c2", "c3"]

Pcset.get(src: Set): Pcset
  Get a pitch class set (the notes of an octave, ignoring octave and order) from a list of note names or interval names, a chroma or a set number. The result has `chroma`, `setNum`, `normalized` and `intervals` (from C). It returns an empty set (`{ empty: true, ... }`) for anything else, and invalid items of a list are ignored.
  Pcset.get(["c", "d", "e"]).chroma; // => "101010000000"
  Pcset.get(["c", "d", "e"]).intervals; // => ["1P", "2M", "3M"]
  Pcset.get("101010000000").intervals; // => ["1P", "2M", "3M"]

Pcset.includes(set: Set): (noteName: NoteName) => boolean
  [deprecated] Use Pcset.isNoteIncludedIn instead.

Pcset.intervals(set: Set): IntervalName[]
  Get the interval names of a pitch class set, measured from C and ordered by pitch class. It returns `[]` for an invalid set.
  Pcset.intervals(["c", "d", "e"]); // => ["1P", "2M", "3M"]
  Pcset.intervals(["D", "F#"]); // => ["2M", "5d"]

Pcset.isChroma(set: any): set is PcsetChroma
  Test if a value is a pitch class set chroma: a string of 12 characters, each "0" or "1".
  Pcset.isChroma("101010000000"); // => true
  Pcset.isChroma("10101"); // => false

Pcset.isEqual(s1: Set, s2: Set): boolean
  Test if two pitch class sets have the same pitch classes, whatever their octaves and order.
  Pcset.isEqual(["c2", "d3"], ["c5", "d2"]); // => true
  Pcset.isEqual(["C", "E"], ["C", "D"]); // => false

Pcset.isNoteIncludedIn(set: Set): (noteName: NoteName) => boolean
  Get a function that tests if a note belongs to a pitch class set, ignoring its octave. It returns false for an invalid note.
  Pcset.isNoteIncludedIn(["C", "E", "G"])("C4"); // => true
  Pcset.isNoteIncludedIn(["C", "E", "G"])("C#4"); // => false

Pcset.isSubsetOf(set: Set): (notes: Set | Pcset) => boolean
  Get a function that tests if a collection of notes is a proper subset of a given set (all its pitch classes are in the set, and the set has more). The same set is not a subset of itself.
  Pcset.isSubsetOf(["C", "E", "G"])(["e6", "c4"]); // => true
  Pcset.isSubsetOf(["C", "E", "G"])(["e6", "c4", "d3"]); // => false
  Pcset.isSubsetOf(["C", "E", "G"])(["C", "E", "G"]); // => false

Pcset.isSupersetOf(set: Set): (notes: Set) => boolean
  Get a function that tests if a collection of notes is a proper superset of a given set (it contains all its pitch classes and at least one more).
  Pcset.isSupersetOf(["C", "E", "G"])(["e6", "a", "c4", "g2"]); // => true
  Pcset.isSupersetOf(["C", "E", "G"])(["c6", "e4", "g3"]); // => false

Pcset.modes(set: Set, normalize?: boolean): PcsetChroma[]
  Get the rotations of the chroma of a pitch class set, each one a chroma having one of its notes as C. By default the rotations that start with "0" (not a note of the set) are left out, so there is one per note of the set.
  Pcset.modes(["C", "D", "E"]); // => ["101010000000", "101000000010", "100000001010"]
  Pcset.modes(["C", "D", "E"]).map(Pcset.intervals); // => [["1P", "2M", "3M"], ["1P", "2M", "7m"], ["1P", "6m", "7m"]]
  Pcset.modes(["C", "D", "E"], false).length; // => 12

Pcset.notes(set: Set): NoteName[]
  Get the note names of a pitch class set, spelled as intervals from C. It returns `[]` for an invalid set. To spell them from another tonic, use `Scale.get` or `Chord.get`.
  Pcset.notes(["c", "d", "e"]); // => ["C", "D", "E"]
  Pcset.notes(["D", "F#"]); // => ["D", "Gb"]

Pcset.num(set: Set): number
  Get the number of a pitch class set: its chroma read as a binary number (from 1 to 4095), or 0 for an invalid set.
  Pcset.num(["c", "d", "e"]); //=> 2688

Pcset.pcset(src: Set): Pcset
  [deprecated] Use Pcset.get instead.

## RhythmPattern

import { RhythmPattern } from "tonal"; // or: import * as RhythmPattern from "@tonaljs/rhythm-pattern";

RhythmPattern.binary(...numbers: number[]): RhythmPattern
  Create a rhythm pattern (an array of 0s and 1s) from the binary digits of one or more numbers, concatenated.
  RhythmPattern.binary(13) // => [1, 1, 0, 1]
  RhythmPattern.binary(12, 13) // => [1, 1, 0, 0, 1, 1, 0, 1]

RhythmPattern.euclid(steps: number, beats: number): RhythmPattern
  Create a Euclidean rhythm pattern: the given number of beats spread as evenly as possible over the steps.
  RhythmPattern.euclid(8, 3) // => [1, 0, 0, 1, 0, 0, 1, 0]

RhythmPattern.hex(hexNumber: string): RhythmPattern
  Create a rhythm pattern from a hexadecimal string: each digit gives four steps (non-hex digits give four 0s).

  Create a rhythm pattern from the gaps between onsets: each number is how many 0s follow a 1.

## Scale

import { Scale } from "tonal"; // or: import * as Scale from "@tonaljs/scale";

Scale.chords(scaleName: string | ScaleNameTokens, options?: Partial<ScaleChordsOptions>): ScaleChord[]
  Get the chord built on each degree of a scale, by stacking scale notes.
  Scale.chords("C major").map((c) => c.symbol); // => ["C", "Dm", "Em", "F", "G", "Am", "Bdim"]
  Scale.chords("C major", { size: 4 })[6]; // => { degree: 7, notes: ["B", "D", "F", "A"], symbol: "Bm7b5" }
  Scale.chords("C4 major")[4].notes; // => ["G4", "B4", "D5"]

Scale.degrees(scaleName: string | ScaleNameTokens): (degree: ScaleStep) => string
  Get a function that returns the note of a scale by its degree, 1 being the tonic (use `Scale.steps` for 0-based). The function returns `""` for degree 0, an invalid degree or a scale without tonic.
  [1, 2, 3].map(Scale.degrees("C major")); // => ["C", "D", "E"]
  [1, 2, 3].map(Scale.degrees("C4 major")); // => ["C4", "D4", "E4"]
  Scale.degrees("C major")("7b"); // => "Bb"

Scale.detect(notes: string[], options?: { tonic?: string; match?: 'exact' | 'fit'; }): string[]
  Find the names of the scales that contain a list of notes, best first. The tonic is the first note, or `options.tonic`. A scale that has exactly the notes comes first. With `match: "exact"` it's the only result (and the list is empty if there is none); by default it's followed by the scales that have the notes and more. It returns `[]` for no valid tonic.
  Scale.detect(["D", "E", "F", "G", "A", "B", "C"], { match: "exact" }); // => ["D dorian"]
  Scale.detect(["C", "D", "E", "F", "G", "A", "B"], { tonic: "D", match: "exact" }); // => ["D dorian"]
  Scale.detect(["C", "D", "E", "G", "A"], { match: "exact" }); // => ["C major pentatonic"]

Scale.extended(name: string): string[]
  Get the names of the scale types that have all the notes of a scale and at least one more (the tonic is ignored). It returns `[]` if the scale isn't valid. The opposite of `Scale.reduced`.
  Scale.extended("major"); // => ["bebop", "bebop major", "ichikosucho", "chromatic"]

Scale.get(src: ScaleName | ScaleNameTokens): Scale
  Get a scale from its name, as an object with `notes`, `intervals`, `name`, `tonic`, `type`, `aliases` and `chroma`. The name is a tonic (with an optional octave) followed by a scale type name or alias, or the tokens `[tonic, type]`. Without a tonic the scale has intervals but no notes. It returns `{ empty: true, ... }` if the scale type isn't in the dictionary.
  Scale.get("C major").notes; // => ["C", "D", "E", "F", "G", "A", "B"]
  Scale.get("C4 major").notes; // => ["C4", "D4", "E4", "F4", "G4", "A4", "B4"]
  Scale.get(["D", "dorian"]).name; // => "D dorian"

Scale.modeNames(name: string): ScaleMode[]
  Get the modes of a scale (its rotations) that have a name in the dictionary, as `[tonic, scale type name]` pairs. The tonics are note names, or intervals if the scale has no tonic. It returns `[]` if the scale isn't valid.
  Scale.modeNames("C pentatonic"); // => [["C", "major pentatonic"], ["D", "egyptian"], ["E", "malkos raga"], ["G", "ritusen"], ["A", "minor pentatonic"]]

Scale.names(): string[]
  Get the names of all the scale types in the dictionary. An alias of `ScaleType.names`.
  Scale.names().slice(0, 3); // => ["major pentatonic", "major", "minor"]

Scale.nearest(scaleName: string | ScaleNameTokens, noteName: string): string
  Find the scale note closest to a note, by pitch.
  Scale.nearest("C major", "F#4"); // => "G4"
  Scale.nearest("C major", "Gb4"); // => "G4"
  Scale.nearest("C major", "Eb"); // => "E"

Scale.outside(scale: string | string[], chord: string | string[]): string[]
  Get the notes of a chord that aren't in a scale: the borrowed or chromatic notes. Each is a pitch class spelled as in the chord, listed once, in the chord order. The result is empty exactly when the chord fits the scale.
  Scale.outside("C major", "Bb7"); // => ["Bb", "Ab"]
  Scale.outside("C major", "Fm"); // => ["Ab"]
  Scale.outside("C major", "Dm7"); // => []

Scale.rangeOf(scale: string | string[]): (fromNote: string, toNote: string) => (string | undefined)[]
  Get a function that lists the scale notes between two notes, both included. The notes are spelled as in the scale. The function takes two note names with octave and returns `[]` if one isn't valid. The scale is a name or a list of notes.
  Scale.rangeOf("C pentatonic")("C4", "C5"); // => ["C4", "D4", "E4", "G4", "A4", "C5"]
  Scale.rangeOf(["C", "D", "E"])("C4", "E5"); // => ["C4", "D4", "E4", "C5", "D5", "E5"]
  Scale.rangeOf("C major")("C4", "x"); // => []

Scale.reduced(name: string): string[]
  Get the names of the scale types whose notes are all in a scale and are fewer (the tonic is ignored). It returns `[]` if the scale isn't valid. The opposite of `Scale.extended`.
  Scale.reduced("major"); // => ["major pentatonic", "ionian pentatonic", "ritusen"]

Scale.scale(src: ScaleName | ScaleNameTokens): Scale
  [deprecated] Use Scale.get instead.

Scale.scaleChords(name: string): string[]
  Get the chord type symbols whose notes are all in a scale, without tonic. It returns `[]` if the scale isn't valid. To get the chord built on each degree, use `Scale.chords`.
  Scale.scaleChords("pentatonic"); // => ["5", "M", "6", "sus2", "Madd9"]

Scale.scaleNotes(notes: NoteName[]): string[]
  Turn a list of notes into a scale: its unique pitch classes sorted from the first note, which is the tonic. Octaves and repetitions are dropped, and invalid notes are ignored.
  Scale.scaleNotes(["C4", "c3", "C5", "C4", "c4"]); // => ["C"]
  Scale.scaleNotes(["D4", "c#5", "A5", "F#6"]); // => ["D", "F#", "A", "C#"]

Scale.steps(scaleName: string | ScaleNameTokens): (step: ScaleStep) => string
  Get a function that returns the note of a scale by its step, 0 being the tonic. It's `Scale.degrees` with 0-based index: negative steps go down, and the accidental syntax is the same. The function returns `""` for an invalid step or a scale without tonic.
  [-1, 0, 1].map(Scale.steps("C4 major")); // => ["B3", "C4", "D4"]
  Scale.steps("C major")("6b"); // => "Bb"

Scale.tokenize(name: ScaleName): ScaleNameTokens
  Split a scale name into `[tonic, type]`. The tonic is a normalized note name (or `""` if there is no tonic) and the type is the rest of the name in lower case, which can be any string: this function doesn't check that the scale type exists. It never throws.
  Scale.tokenize("C mixolydian"); // => ["C", "mixolydian"]
  Scale.tokenize("Cx major"); // => ["C##", "major"]
  Scale.tokenize("anything is valid"); // => ["", "anything is valid"]

Scale.transposeBySteps(scaleName: string | ScaleNameTokens, noteName: string, steps: number): string
  Transpose a note of the scale by a number of scale steps.
  Scale.transposeBySteps("C major", "E4", 2); // => "G4"
  Scale.transposeBySteps("C major", "E4", -3); // => "B3"
  Scale.transposeBySteps("C major", "E", 2); // => "G"

## TimeSignature

import { TimeSignature } from "tonal"; // or: import * as TimeSignature from "@tonaljs/time-signature";

TimeSignature.get(literal: TimeSignatureLiteral): TimeSignature
  Get the TimeSignature object of a name ("3/4", "3+2/8", or the symbols "C" and "C|"), or of a `[upper, lower]` pair. `type` is "simple", "compound", "irregular" or "irrational". Invalid input gives an object with `empty: true` and `name: ""`.

TimeSignature.parse(literal: TimeSignatureLiteral): ParsedTimeSignature
  Parse a time signature into `[upper, lower]`: the upper is a number, or a list of numbers for an additive signature. Invalid input gives `[0, 0]`. Use `TimeSignature.get` to get a validated object.
  TimeSignature.parse("3/4") // => [3, 4]
  TimeSignature.parse("3+2/8") // => [[3, 2], 8]
  TimeSignature.parse("x") // => [0, 0]

## VoiceLeading

import { VoiceLeading } from "tonal"; // or: import * as VoiceLeading from "@tonaljs/voice-leading";

VoiceLeading.distance(from: string[], to: string[], options?: VoiceLeadingOptions): number | null
  Get the size of the smallest voice leading between two voicings (lists of notes with octave): by default, the total number of semitones the voices move. It returns `null` if a voicing is empty or has a note without octave. See `VoiceLeading.get` for how the voices are matched.
  VoiceLeading.distance(["C4", "E4", "G4"], ["B3", "D4", "G4"]); // => 3
  VoiceLeading.distance(["C4", "E4", "G4"], ["B3", "D4", "G4"], { metric: "Linf" }); // => 2
  VoiceLeading.distance(["C4", "E4"], ["C"]); // => null

VoiceLeading.get(from: string[], to: string[], options?: VoiceLeadingOptions): VoiceLeadingMapping
  Get the smallest voice leading between two voicings (lists of notes with octave): how far the voices move, and where each voice goes. It returns `{ distance: null, moves: [] }` if a voicing is empty or has a note without octave.
  VoiceLeading.get(["C4", "E4", "G4"], ["B3", "D4", "G4"]); // => { distance: 3, moves: [["C4", "B3"], ["E4", "D4"], ["G4", "G4"]] }
  VoiceLeading.get(["C4", "E4", "G4"], ["C4", "E4", "G4", "B4"]).moves; // => [["C4", "C4"], ["E4", "E4"], ["G4", "G4"], ["G4", "B4"]]

VoiceLeading.path(candidates: string[][][], options?: VoiceLeadingPathOptions): VoiceLeadingPath
  Get the smoothest path through the candidate voicings of a list of chords: one voicing per chord, so that the total voice-leading distance between consecutive voicings (plus the cost of each candidate, if given) is the smallest possible. It returns `[]` for a chord without valid candidates and leads the next chord from the last voicing.
  VoiceLeading.path([[["C4", "E4", "G4"]], [["F4", "A4", "C5"], ["C4", "F4", "A4"]], [["G4", "B4", "D5"], ["B3", "D4", "G4"]]]).voicings; // => [["C4", "E4", "G4"], ["C4", "F4", "A4"], ["B3", "D4", "G4"]]
  VoiceLeading.path([[["C4", "E4", "G4"]], [["F4", "A4", "C5"], ["C4", "F4", "A4"]], [["G4", "B4", "D5"], ["B3", "D4", "G4"]]]).steps; // => [0, 3, 6]

VoiceLeading.pcDistance(from: string[], to: string[], options?: VoiceLeadingOptions): number | null
  Get the size of the smallest voice leading between two sets of pitch classes, without octaves: each voice moves the shortest way to its new pitch class. Notes with octave are read as their pitch class, and repeated pitch classes count once. It returns `null` if a list is empty or has an invalid note.
  VoiceLeading.pcDistance(["C", "E", "G"], ["C", "E", "A"]); // => 2
  VoiceLeading.pcDistance(Chord.notes("C"), Chord.notes("Em")); // => 1
  VoiceLeading.pcDistance(Chord.notes("Dm7"), Chord.notes("G7")); // => 3

VoiceLeading.topNoteDiff(voicings: string[][], lastVoicing: string[]): string[]
  [deprecated] It only looks at the top note: use `VoiceLeading.path`, or `Voicing.get` with `previous`, which measure every voice. Removed in tonal 7.
  Pick the voicing whose top note is closest to the top note of the last voicing (the first of them on a tie). If there is no last voicing, it picks the first one. It returns `[]` if there are no voicings, and doesn't change the given list.
  VoiceLeading.topNoteDiff([["C4", "E4", "G4"], ["E4", "G4", "C5"], ["G4", "C5", "E5"]], ["D4", "F4", "A4"]); // => ["C4", "E4", "G4"]

## Voicing

import { Voicing } from "tonal"; // or: import * as Voicing from "@tonaljs/voicing";

Voicing.analyze(notes: string[], chord?: string): VoicingAnalysis
  Describe a voicing (a list of notes with octave): its notes from the bottom, the bass and top notes, the span, the spacing in chord steps and its name (block, close, open, drop2, drop3, drop24, spread), and, with a chord symbol, the chord degree of each note. It returns `empty: true` if the voicing is empty or has a note without octave.
  Voicing.analyze(["G3", "C4", "E4", "B4"], "Cmaj7").spacing; // => "drop2"
  Voicing.analyze(["G3", "C4", "E4", "B4"], "Cmaj7").degrees; // => ["5", "1", "3", "7"]
  Voicing.analyze(["C4", "G4", "E5"]).steps; // => [2, 2]

Voicing.drop(notes: string[], n: number | number[]): string[]
  Drop voices of a voicing an octave: `n` is the position of the voice counting from the top (1 is the top voice), or a list of positions. A close voicing with the 2nd voice from the top dropped is a "drop 2" voicing. The result is sorted from the bottom, and the spelling is kept. It returns `[]` if the voicing is invalid or a position doesn't exist.
  Voicing.drop(["C4", "E4", "G4", "B4"], 2); // => ["G3", "C4", "E4", "B4"]
  Voicing.drop(["C4", "E4", "G4", "B4"], 3); // => ["E3", "C4", "G4", "B4"]
  Voicing.drop(["C4", "E4", "G4", "B4"], [2, 4]); // => ["C3", "G3", "E4", "B4"]

Voicing.get(chord: string, options?: VoicingOptions): string[]
Voicing.get(chord: string, range?: string[], dictionary?: VoicingDictionary, voiceLeading?: VoiceLeadingFunction, lastVoicing?: string[]): string[]  [deprecated] Use the options object: `Voicing.get(chord, { range, previous })`. Removed in tonal 7.
  Get a voicing of a chord: a list of notes with octave, from the bottom. By default it has the notes of the symbol, in root position, near the middle of the range ["C3", "C5"]: `Voicing.get("C7")` is C, E, G and Bb. Options choose the number of voices, the tensions, who plays the bass, the range, the top note, and lead from a previous voicing. It returns `[]` if the chord is not valid or no voicing fits.
  Voicing.get("C7"); // => ["C4", "E4", "G4", "Bb4"]
  Voicing.get("Dm7", { voices: 4, extensions: "available", bass: "external" }); // => ["F3", "A3", "C4", "E4"]
  Voicing.get("G7", { voices: 4, extensions: "available", bass: "external", previous: ["F3", "A3", "C4", "E4"] }); // => ["F3", "A3", "B3", "E4"]

Voicing.invert(notes: string[], n: number): string[]
  Invert a voicing: move every voice `n` steps along the voicing's own notes (up for a positive `n`, down for a negative one), so the spacing doesn't change. In close position it's the usual inversion (C4 E4 G4 becomes E4 G4 C5); in other spacings the voicing keeps its shape. The spelling is kept. It returns `[]` if the voicing is empty or has a note without octave.
  Voicing.invert(["C4", "E4", "G4"], 1); // => ["E4", "G4", "C5"]
  Voicing.invert(["C4", "G4", "E5"], 1); // => ["E4", "C5", "G5"]
  Voicing.invert(["C4", "E4", "G4"], -1); // => ["G3", "C4", "E4"]

Voicing.search(chord: string, options?: VoicingOptions): string[][]
Voicing.search(chord: string, range?: string[], dictionary?: VoicingDictionary): string[][]  [deprecated] Use the options object: `Voicing.search(chord, { range })`. Removed in tonal 7.
  Get every voicing of a chord that fits the options, best first. Each voicing is a list of notes with octave, from the bottom. It returns `[]` if the chord is not valid or no voicing fits.
  Voicing.search("C").slice(0, 2); // => [["C4", "E4", "G4"], ["C3", "E3", "G3"]]
  Voicing.search("C7", { voices: 3 })[0]; // => ["C4", "E4", "Bb4"]
  Voicing.search("C7", { bass: "root", voices: 3 })[0]; // => ["C2", "Bb3", "E4", "G4"]

Voicing.sequence(chords: (string | VoicingChord)[], options?: SequenceOptions): string[][]
Voicing.sequence(chords: string[], range?: string[], dictionary?: VoicingDictionary, voiceLeading?: VoiceLeadingFunction, lastVoicing?: string[]): string[][]  [deprecated] Use the options object: `Voicing.sequence(chords, { range, previous })`. Removed in tonal 7.
  Get a voicing for each chord of a list: the smoothest path through the voicings `Voicing.search` gives for each chord, so that the voices move as little as possible over the whole progression (an exact path, not a greedy one). The options are the ones of `Voicing.get`, for every chord, and each chord can be an object with its own (`{ chord: "G7", voices: 5 }`), all but `previous` and `metric`, which belong to the whole sequence. `maxLeap` limits the move of each voice. A chord without voicings gives `[]`, and the next one is led from the last.
  Voicing.sequence(["C", "Am", "F", "G"]); // => [["C4", "E4", "G4"], ["A3", "C4", "E4"], ["F3", "A3", "C4"], ["G3", "B3", "D4"]]
  Voicing.sequence(["Dm7", "G7", "Cmaj7"], { voices: 4, extensions: "available", bass: "external" }); // => [["F3", "A3", "C4", "E4"], ["F3", "A3", "B3", "E4"], ["E3", "G3", "B3", "D4"]]
  Voicing.sequence(["Dm7", { chord: "G7", voices: 5 }, "Cmaj7"], { voices: 4 }).map((v) => v.length); // => [4, 5, 4]

Voicing.templates(): string[]
  Get the names of tonal's voicing templates, for the `template` option: shell, guide-tones, rootless, rootless-spread, so-what, fourths, upper-structure (and upper-structure-II, -bIII, -bV, -bVI, -VI) and triad-over-root. Each one is in two or more jazz piano books (see the README).
  Voicing.templates().includes("rootless"); // => true
